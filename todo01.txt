## Final unified architecture

### System 1: `MemoryLongTerm` — semantic memory (facts only)

Purpose: Permanent facts about user  
Storage: `MemoryLongTerm` table  
Categories: `fact`, `preference`, `interest`, `relationship`, `context`, `achievement`

### System 2: `StyleAnchors` — procedural memory (behavioral patterns)

Purpose: How user communicates  
Storage: `style_anchors` table (enhanced)  
Types: `interaction`, `phrase`, `pattern`

### System 3: `StyleVector` — style configuration

Purpose: Numeric style parameters  
Storage: `Twin.styleVector` (JSONB) — keep as is

### System 4: `MemorySession` — episodic memory

Purpose: Chat summaries  
Storage: `MemorySession` table — keep as is

### Remove: `mem_chunks`

Action: Migrate data, then deprecate

```javascript
const { Pool } = require('pg');
const fs = require('fs');
require('dotenv').config();

const pool = new Pool({
  connectionString: process.env.DATABASE_URL,
});

async function migrateData() {
  const client = await pool.connect();
  
  try {
    console.log('🔄 Starting migration...');
    
    // 1. Migrate mem_chunks.facts → MemoryLongTerm
    console.log('1️⃣ Migrating facts to MemoryLongTerm...');
    const factsResult = await client.query(`
      SELECT DISTINCT ON (twin_id, text) 
        twin_id, text, ts
      FROM mem_chunks
      WHERE bucket = 'facts'
      ORDER BY twin_id, text, ts DESC
    `);
    
    for (const row of factsResult.rows) {
      const key = `fact_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
      await client.query(`
        INSERT INTO "MemoryLongTerm" (id, "twinId", key, value, category, source, "createdAt")
        VALUES ($1, $2, $3, $4, $5, $6, $7)
        ON CONFLICT ("twinId", key) DO NOTHING
      `, [key, row.twin_id, key, row.text, 'fact', 'migration', row.ts]);
    }
    console.log(`   ✅ Migrated ${factsResult.rows.length} facts`);
    
    // 2. Migrate mem_chunks.voice → style_anchors (as phrases)
    console.log('2️⃣ Migrating voice patterns to style_anchors...');
    const voiceResult = await client.query(`
      SELECT DISTINCT ON (twin_id, text) 
        twin_id, text, ts
      FROM mem_chunks
      WHERE bucket = 'voice'
      ORDER BY twin_id, text, ts DESC
    `);
    
    for (const row of voiceResult.rows) {
      const id = `anchor_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
      await client.query(`
        INSERT INTO "style_anchors" (id, twin_id, type, phrase, tags, created_at)
        VALUES ($1, $2, $3, $4, $5, $6)
        ON CONFLICT DO NOTHING
      `, [id, row.twin_id, 'phrase', row.text, ARRAY['migrated'], row.ts]);
    }
    console.log(`   ✅ Migrated ${voiceResult.rows.length} voice patterns`);
    
    console.log('✅ Migration completed successfully!');
    
  } catch (error) {
    console.error('❌ Migration failed:', error);
    throw error;
  } finally {
    client.release();
    await pool.end();
  }
}

migrateData();
```

```typescript
  /**
   * Store style pattern (phrase or pattern)
   * Converts voice patterns to style anchors
   */
  async storeStylePattern(
    twinId: string,
    pattern: string,
    type: 'phrase' | 'pattern' = 'phrase',
    patternType?: 'greeting' | 'agreement' | 'disagreement' | 'question',
    context?: string
  ): Promise<void> {
    try {
      const { styleAnchorsQueries } = await import('../config/database');
      
      if (type === 'phrase') {
        // Store as phrase
        await styleAnchorsQueries.create(
          twinId,
          '', // No user utterance for phrases
          '', // No ideal reply for phrases
          ['phrase', 'auto'] // Tags
        );
        
        // Update with phrase data
        await db.query(`
          UPDATE "style_anchors"
          SET type = $1, phrase = $2, context = $3
          WHERE twin_id = $4 AND type = $1
          ORDER BY created_at DESC
          LIMIT 1
        `, [type, pattern, context || null, twinId]);
      } else {
        // Store as pattern
        await styleAnchorsQueries.create(
          twinId,
          pattern, // Pattern examples go here
          '', // No ideal reply needed
          ['pattern', 'auto']
        );
        
        await db.query(`
          UPDATE "style_anchors"
          SET type = $1, pattern_type = $2, phrase = $3, context = $4
          WHERE twin_id = $5 AND type = $1
          ORDER BY created_at DESC
          LIMIT 1
        `, [type, patternType || null, pattern, context || null, twinId]);
      }
      
      logger.info(`Stored style pattern for twin ${twinId}: ${pattern} (${type})`);
    } catch (error) {
      logger.error('Error storing style pattern:', error);
    }
  }

  /**
   * Get relevant style patterns for a query
   */
  async getRelevantStylePatterns(
    twinId: string,
    userMessage: string,
    limit: number = 3
  ): Promise<Array<{
    type: string;
    phrase?: string;
    userUtterance?: string;
    idealReply?: string;
    patternType?: string;
  }>> {
    try {
      const { styleAnchorsQueries } = await import('../config/database');
      
      // Get most similar interaction anchors
      const interactionAnchors = await styleAnchorsQueries.findByTwinAndSimilarity(
        twinId,
        userMessage,
        Math.min(limit, 2)
      );
      
      // Get common phrases (top 2)
      const phraseResult = await db.query(`
        SELECT phrase, context, type
        FROM "style_anchors"
        WHERE twin_id = $1 AND type = 'phrase'
        ORDER BY created_at DESC
        LIMIT 2
      `, [twinId]);
      
      const results = [
        ...interactionAnchors.map(a => ({
          type: 'interaction',
          userUtterance: a.user_utterance,
          idealReply: a.ideal_reply
        })),
        ...phraseResult.rows.map(p => ({
          type: 'phrase',
          phrase: p.phrase,
          context: p.context
        }))
      ];
      
      return results.slice(0, limit);
    } catch (error) {
      logger.error('Error getting style patterns:', error);
      return [];
    }
  }
```

```typescript
import { Request, Response } from 'express';
import { db } from '../../config/database';
import { memoryService } from '../../services/memoryService';
import { logger } from '../../config/logger';

/**
 * Get all long-term memories for a twin
 */
export const getLongTermMemories = async (req: any, res: Response) => {
  try {
    const { id: twinId } = req.params;
    const { category, limit = 20 } = req.query;
    const userId = req.user.id;
    
    // Verify ownership
    const twinResult = await db.query(`
      SELECT id FROM "Twin" WHERE id = $1 AND "userId" = $2
    `, [twinId, userId]);
    
    if (twinResult.rows.length === 0) {
      return res.status(404).json({ success: false, error: 'Twin not found' });
    }
    
    const memories = await memoryService.getLongTermMemories(
      twinId,
      category as string | undefined,
      parseInt(limit as string)
    );
    
    res.json({ success: true, memories });
  } catch (error) {
    logger.error('Get long-term memories error:', error);
    res.status(500).json({ success: false, error: 'Failed to get memories' });
  }
};

/**
 * Add long-term memory
 */
export const addLongTermMemory = async (req: any, res: Response) => {
  try {
    const { id: twinId } = req.params;
    const { key, value, category = 'fact' } = req.body;
    const userId = req.user.id;
    
    // Verify ownership
    const twinResult = await db.query(`
      SELECT id FROM "Twin" WHERE id = $1 AND "userId" = $2
    `, [twinId, userId]);
    
    if (twinResult.rows.length === 0) {
      return res.status(404).json({ success: false, error: 'Twin not found' });
    }
    
    // Auto-generate key if not provided
    const finalKey = key || `fact_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
    
    await memoryService.storeLongTermMemory(
      twinId,
      finalKey,
      value,
      category,
      'manual'
    );
    
    res.json({ 
      success: true, 
      message: 'Memory stored successfully' 
    });
  } catch (error) {
    logger.error('Add long-term memory error:', error);
    res.status(500).json({ success: false, error: 'Failed to store memory' });
  }
};

/**
 * Update long-term memory
 */
export const updateLongTermMemory = async (req: any, res: Response) => {
  try {
    const { id: twinId, key } = req.params;
```
