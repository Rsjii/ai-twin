```typescript
import { config } from '../config/env';
import { logger } from '../config/logger';

export interface LLMMessage {
  role: 'system' | 'user' | 'assistant';
  content: string;
}

export interface LLMResponse {
  content: string;
  model: string;
  tokensUsed?: number;
}

export class LLMClient {
  private groqApiKey: string | null;
  private openaiApiKey: string | null;
  private useGroq: boolean;

  constructor() {
    this.groqApiKey = process.env.GROQ_API_KEY || null;
    this.openaiApiKey = config.openaiApiKey || null;
    this.useGroq = !!this.groqApiKey; // Prefer Groq if available
    
    if (!this.groqApiKey && !this.openaiApiKey) {
      logger.warn('No LLM API key configured. Set GROQ_API_KEY or OPENAI_API_KEY');
    }
  }

  async generateResponse(
    messages: LLMMessage[],
    options: {
      model?: string;
      maxTokens?: number;
      temperature?: number;
      responseFormat?: { type: 'json_object' };
    } = {}
  ): Promise<LLMResponse> {
    if (this.useGroq) {
      return this.callGroq(messages, options);
    } else if (this.openaiApiKey) {
      return this.callOpenAI(messages, options);
    } else {
      throw new Error('No LLM API key configured');
    }
  }

  private async callGroq(
    messages: LLMMessage[],
    options: any
  ): Promise<LLMResponse> {
    try {
      const response = await fetch('https://api.groq.com/openai/v1/chat/completions', {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${this.groqApiKey}`,
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          model: options.model || 'mixtral-8x7b-32768',
          messages: messages.map(m => ({ role: m.role, content: m.content })),
          max_tokens: options.maxTokens || 512,
          temperature: options.temperature || 0.7,
          ...(options.responseFormat || {})
        })
      });

      if (!response.ok) {
        const errorText = await response.text();
        throw new Error(`Groq API error: ${response.status} ${errorText}`);
      }

      const data = await response.json();
      return {
        content: data.choices[0]?.message?.content?.trim() || '',
        model: 'mixtral-8x7b',
        tokensUsed: data.usage?.total_tokens
      };
    } catch (error) {
      logger.error('Groq API failed, falling back to OpenAI:', error);
      if (this.openaiApiKey) {
        return this.callOpenAI(messages, options);
      }
      throw error;
    }
  }

  private async callOpenAI(
    messages: LLMMessage[],
    options: any
  ): Promise<LLMResponse> {
    const { default: OpenAI } = await import('openai');
    const openai = new OpenAI({ apiKey: this.openaiApiKey! });

    const completion = await openai.chat.completions.create({
      model: options.model || 'gpt-4o-mini',
      messages: messages.map(m => ({ role: m.role, content: m.content })),
      max_tokens: options.maxTokens || 512,
      temperature: options.temperature || 0.7,
      ...(options.responseFormat || {})
    });

    return {
      content: completion.choices[0]?.message?.content?.trim() || '',
      model: 'gpt-4o-mini',
      tokensUsed: completion.usage?.total_tokens
    };
  }
}

export const llmClient = new LLMClient();
```

```typescript
import { logger } from '../config/logger';
import crypto from 'crypto';

interface CacheEntry {
  response: string;
  timestamp: number;
  twinId: string;
}

class ResponseCache {
  private cache: Map<string, CacheEntry> = new Map();
  private readonly TTL = 60 * 60 * 1000; // 1 hour
  private readonly MAX_ENTRIES_PER_TWIN = 10;

  /**
   * Generate cache key from twinId and message
   */
  private generateKey(twinId: string, message: string): string {
    const hash = crypto.createHash('md5').update(message.toLowerCase().trim()).digest('hex');
    return `${twinId}:${hash}`;
  }

  /**
   * Get cached response if available and not expired
   */
  get(twinId: string, message: string): string | null {
    const key = this.generateKey(twinId, message);
    const entry = this.cache.get(key);

    if (!entry) {
      return null;
    }

    // Check if expired
    const age = Date.now() - entry.timestamp;
    if (age > this.TTL) {
      this.cache.delete(key);
      return null;
    }

    logger.debug(`Cache hit for twin ${twinId}`);
    return entry.response;
  }

  /**
   * Store response in cache
   */
  set(twinId: string, message: string, response: string): void {
    const key = this.generateKey(twinId, message);
    
    // Clean up old entries for this twin
    this.cleanupTwinEntries(twinId);

    this.cache.set(key, {
      response,
      timestamp: Date.now(),
      twinId
    });

    logger.debug(`Cached response for twin ${twinId}`);
  }

  /**
   * Clean up old entries for a specific twin
   */
  private cleanupTwinEntries(twinId: string): void {
    const twinEntries: Array<[string, CacheEntry]> = [];
    
    for (const [key, entry] of this.cache.entries()) {
      if (entry.twinId === twinId) {
        twinEntries.push([key, entry]);
      }
    }

    // Sort by timestamp (oldest first)
    twinEntries.sort((a, b) => a[1].timestamp - b[1].timestamp);

    // Remove oldest entries if over limit
    if (twinEntries.length >= this.MAX_ENTRIES_PER_TWIN) {
      const toRemove = twinEntries.length - this.MAX_ENTRIES_PER_TWIN + 1;
      for (let i = 0; i < toRemove; i++) {
        this.cache.delete(twinEntries[i][0]);
      }
    }
  }

  /**
   * Clear all cache entries
   */
  clear(): void {
    this.cache.clear();
    logger.info('Response cache cleared');
  }

  /**
   * Get cache statistics
   */
  getStats(): { size: number; entries: number } {
    return {
      size: this.cache.size,
      entries: Array.from(this.cache.values()).length
    };
  }
}

export const responseCache = new ResponseCache();
```

```typescript
async function preWarmGroqAPI(): Promise<void> {

      method: 'POST',
      headers: {
        'Authorization': `Bearer ${groqApiKey}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        model: 'mixtral-8x7b-32768',
        messages: [{ role: 'user', content: 'Ping' }],
        max_tokens: 5
      })
    });

    if (response.ok) {
      logger.info('✅ Groq API pre-warmed successfully');
    } else {
      logger.warn('⚠️ Groq API pre-warm failed:', response.statusText);
    }
  } catch (error) {
    logger.warn('⚠️ Groq API pre-warm error:', error);
    // Don't fail startup if pre-warm fails
  }
}
```
