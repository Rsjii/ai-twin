### 1. Settings -> Privacy & Instructions improvements

#### a. Save button color change on edit
Current state: Button static color hai, changes ka indication nahi.

Solution:
- Initial state: Gray/disabled look
- On any toggle change: Active color (blue/green gradient)
- Track original values vs current values
- Reset to original after save

Implementation:
```javascript
// Track original settings
let originalSettings = {
  showChatHistory: document.getElementById('showChatHistory')?.checked ?? true,
  requireLogin: document.getElementById('requireLogin')?.checked ?? false,
  allowLikes: document.getElementById('allowLikes')?.checked ?? true,
  allowFollows: document.getElementById('allowFollows')?.checked ?? true,
  allowShares: document.getElementById('allowShares')?.checked ?? true,
};

// Check if settings changed
function hasSettingsChanged() {
  return (
    (document.getElementById('showChatHistory')?.checked ?? true) !== originalSettings.showChatHistory ||
    (document.getElementById('requireLogin')?.checked ?? false) !== originalSettings.requireLogin ||
    (document.getElementById('allowLikes')?.checked ?? true) !== originalSettings.allowLikes ||
    (document.getElementById('allowFollows')?.checked ?? true) !== originalSettings.allowFollows ||
    (document.getElementById('allowShares')?.checked ?? true) !== originalSettings.allowShares
  );
}

// Update button style based on changes
function updateSaveButtonStyle() {
  const btn = document.getElementById('savePrivacySettingsBtn');
  if (hasSettingsChanged()) {
    btn.classList.remove('bg-gray-400', 'cursor-not-allowed');
    btn.classList.add('bg-gradient-to-r', 'from-primary', 'to-accent', 'cursor-pointer', 'hover:shadow-lg');
  } else {
    btn.classList.remove('bg-gradient-to-r', 'from-primary', 'to-accent', 'hover:shadow-lg');
    btn.classList.add('bg-gray-400', 'cursor-not-allowed');
  }
}

// Add listeners to all toggles
document.querySelectorAll('.privacy-toggle').forEach(toggle => {
  toggle.addEventListener('change', updateSaveButtonStyle);
});
```

```typescript
// Get full chat history with messages
export const getFullChatHistory = async (req: any, res: Response) => {
  const { twinId } = req.params;
  const userId = req.user.id;
  const { page = 1, limit = 20 } = req.query;

  // Get chats with message counts
  const chats = await db.query(`
    SELECT 
      c.id,
      c."createdAt",
      c."lastMessage",
      COUNT(m.id) as message_count,
      MAX(m."createdAt") as last_message_time
    FROM "Chat" c
    LEFT JOIN "Message" m ON c.id = m."chatId"
    WHERE c."twinId" = $1 AND c."userId" = $2
    GROUP BY c.id, c."createdAt", c."lastMessage"
    ORDER BY c."createdAt" DESC
    LIMIT $3 OFFSET $4
  `, [twinId, userId, limit, (page - 1) * limit]);

  // For each chat, get messages
  const chatsWithMessages = await Promise.all(
    chats.rows.map(async (chat) => {
      const messages = await db.query(`
        SELECT id, sender, content, "createdAt", approved
        FROM "Message"
        WHERE "chatId" = $1
        ORDER BY "createdAt" ASC
      `, [chat.id]);

      return {
        ...chat,
        messages: messages.rows
      };
    })
  );

  res.json({ success: true, chats: chatsWithMessages });
};
```
