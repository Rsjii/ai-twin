/**
 * Get unified memory statistics
 * GET /api/memory/:id/memory/stats (NEW VERSION)
 */
export const getUnifiedMemoryStats = async (req: any, res: Response) => {
  try {
    const { id: twinId } = req.params;
    const userId = req.user?.id || req.userId;
    
    // Verify ownership
    const twinResult = await db.query(
      'SELECT id FROM "Twin" WHERE id = $1 AND "userId" = $2',
      [twinId, userId]
    );
    
    if (!twinResult || twinResult.rows.length === 0) {
      throw createError.notFound('Twin not found');
    }
    
    // Get MemoryLongTerm stats
    const longTermResult = await db.query(`
      SELECT 
        category,
        COUNT(*) as count
      FROM "MemoryLongTerm"
      WHERE "twinId" = $1
      GROUP BY category
    `, [twinId]);
    
    // Get StyleAnchors stats
    const anchorsResult = await db.query(`
      SELECT 
        type,
        COUNT(*) as count
      FROM "style_anchors"
      WHERE twin_id = $1
      GROUP BY type
    `, [twinId]);
    
    const totalMemories = longTermResult.rows.reduce((sum, row) => sum + parseInt(row.count), 0);
    const totalAnchors = anchorsResult.rows.reduce((sum, row) => sum + parseInt(row.count), 0);
    
    res.json({
      success: true,
      total: totalMemories + totalAnchors,
      longTermMemories: {
        total: totalMemories,
        byCategory: longTermResult.rows.map(row => ({
          category: row.category,
          count: parseInt(row.count)
        }))
      },
      styleAnchors: {
        total: totalAnchors,
        byType: anchorsResult.rows.map(row => ({
          type: row.type,
          count: parseInt(row.count)
        }))
      }
    });
  } catch (error) {
    throw createError.internal('Failed to get memory statistics', error);
  }
};
```

/**
 * Retrieve memories (UNIFIED VERSION)
 * GET /api/memory/:id/memory/retrieve
 */
export const retrieveUnifiedMemories = async (req: any, res: Response) => {
  try {
    const { id: twinId } = req.params;
    const { bucket, limit = 10, offset = 0 } = req.query;
    const userId = req.user?.id || req.userId;
    
    // Verify ownership
    const twinResult = await db.query(
      'SELECT id FROM "Twin" WHERE id = $1 AND "userId" = $2',
      [twinId, userId]
    );
    
    if (!twinResult || twinResult.rows.length === 0) {
      throw createError.notFound('Twin not found');
    }
    
    if (bucket === 'facts') {
      // Get from MemoryLongTerm
      const { getLongTermMemories } = await import('../twin/longTermMemoryController');
      req.query.category = 'fact';
      req.query.limit = limit;
      req.query.offset = offset;
      return getLongTermMemories(req, res);
      
    } else if (bucket === 'voice') {
      // Get from StyleAnchors (phrases)
      const { getTwinPhrases } = await import('../twin/styleAnchorController');
      req.query.limit = limit;
      return getTwinPhrases(req, res);
      
    } else if (bucket === 'all') {
      // Get from both
      const [longTermMemories, styleAnchors] = await Promise.all([
        db.query(`
          SELECT key, value, category, "createdAt", "updatedAt"
          FROM "MemoryLongTerm"
          WHERE "twinId" = $1
          ORDER BY "updatedAt" DESC
          LIMIT $2 OFFSET $3
        `, [twinId, parseInt(limit as string), parseInt(offset as string)]),
        db.query(`
          SELECT id, phrase, user_utterance, ideal_reply, tags, created_at
          FROM "style_anchors"
          WHERE twin_id = $1 AND type = 'phrase'
          ORDER BY created_at DESC
          LIMIT $2 OFFSET $3
        `, [twinId, parseInt(limit as string), parseInt(offset as string)])
      ]);
      
      res.json({
        success: true,
        memories: [
          ...longTermMemories.rows.map(row => ({
            type: 'fact',
            key: row.key,
            value: row.value,
            category: row.category,
            createdAt: row.createdAt
          })),
          ...styleAnchors.rows.map(row => ({
            type: 'voice',
            id: row.id,
            phrase: row.phrase,
            userUtterance: row.user_utterance,
            idealReply: row.ideal_reply,
            createdAt: row.created_at
          }))
        ]
      });
    } else {
      throw createError.validation('Invalid bucket. Use "facts", "voice", or "all"');
    }
  } catch (error) {
    throw createError.internal('Failed to retrieve memories', error);
  }
};
```
