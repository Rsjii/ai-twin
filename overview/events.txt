Recent events display code:
```544:571:frontend/src/views/admin-analytics.ejs
// Update recent events (with meta tags)
const eventsContainer = document.getElementById('recentEvents');
if (activity.recentEvents && activity.recentEvents.length > 0) {
    eventsContainer.innerHTML = activity.recentEvents.slice(0, 3).map(event => {
        // Extract meta tags (simple client-side extraction)
        const meta = event.meta || {};
        const tags = [];
        if (meta.wv) tags.push(`wv:${meta.wv}`);
        if (meta.source) tags.push(`src:${meta.source}`);
        if (meta.twinId) tags.push(`twin:${meta.twinId.substring(0, 6)}`);
        
        const tagsHtml = tags.length > 0 
            ? `<div class="flex flex-wrap gap-1 mt-1">${tags.map(tag => 
                `<span class="px-1.5 py-0.5 text-xs bg-blue-100 text-blue-700 rounded">${tag}</span>`
              ).join('')}</div>`
            : '';
        
        return `
            <div class="text-sm">
                <p class="font-medium text-gray-900">${event.type}</p>
                <p class="text-gray-600 timestamp-display" data-timestamp="${event.createdAt}" data-format="date"></p>
                ${tagsHtml}
            </div>
        `;
    }).join('');
} else {
    eventsContainer.innerHTML = '<p class="text-gray-500 text-sm">No recent events</p>';
}
```
