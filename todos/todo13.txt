## Solution architecture

### Step 1: Fix layout.ejs authentication logic

Current issue:
```ejs
<% if (typeof user !== 'undefined' && user) { %>
```

Problem: `user` might be `{}` or `null` but still truthy.

Fix:
```ejs
<% const isAuthenticated = typeof user !== 'undefined' && user && user.id; %>
<% if (isAuthenticated) { %>
  <!-- Logged in UI -->
<% } else { %>
  <!-- Not logged in UI -->
<% } %>
```

### Step 2: Ensure all controllers pass user

Pattern for all controllers:
```typescript
export function getPage(req: any, res: Response) {
  res.render('page-name', {
    title: 'Page Title',
    user: req.user || null,  // Always pass, even if null
    csrfToken: res.locals['csrfToken']
  });
}
```

### Step 3: Migrate pages to use layout.ejs

Pattern:
```ejs
<%- include('layout', { 
  title: 'Page Title', 
  user: typeof user !== 'undefined' ? user : null,
  body: `
    <!-- Page content here -->
  `
}) %>
```

### Example 2: Controller pattern

```typescript
// Standard pattern for all controllers
export function getPage(req: any, res: Response) {
  // Always pass user, even if null
  res.render('page-name', {
    title: 'Page Title - AI Twin',
    user: req.user || null,
    csrfToken: res.locals['csrfToken']
  });
}

// For pages that need additional data
export async function getPageWithData(req: any, res: Response) {
  try {
    // Fetch additional data
    const data = await fetchData();
    
    res.render('page-name', {
      title: 'Page Title - AI Twin',
      user: req.user || null,  // Always include
      data: data,
      csrfToken: res.locals['csrfToken']
    });
  } catch (error) {
    // Error handling
  }
}
```
