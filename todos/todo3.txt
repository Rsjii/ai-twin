```typescript
// In promptBuilder.ts - add new method
private async getFeedbackContext(twinId: string): Promise<string> {
  // Get recent feedback (last 7 days)
  const feedbackResult = await db.query(`
    SELECT knob, AVG(delta) as avg_delta, COUNT(*) as count
    FROM style_corrections
    WHERE twin_id = $1 AND ts >= NOW() - INTERVAL '7 days'
    GROUP BY knob
    HAVING COUNT(*) >= 2
  `, [twinId]);
  
  if (feedbackResult.rows.length === 0) return '';
  
  const adjustments = feedbackResult.rows.map(f => {
    if (f.avg_delta > 0) return `Increase ${f.knob}`;
    if (f.avg_delta < 0) return `Decrease ${f.knob}`;
  }).filter(Boolean);
  
  return adjustments.length > 0 
    ? `\n## USER FEEDBACK ADJUSTMENTS:\n${adjustments.join('\n')}\n`
    : '';
}
```

Impact: Immediate style improvement based on user feedback  
Effort: Low (1-2 hours)

```typescript
// In memoryService.ts - update getRelevantLongTermMemories
async getRelevantLongTermMemories(twinId: string, userMessage: string, limit: number = 10) {
  // Get memories with relevance scores
  const memories = await db.query(`
    SELECT 
      ml.key, ml.value, ml.category,
      COALESCE(mr."relevanceScore", 0.5) as relevance,
      COALESCE(mr."hitCount", 0) as hits
    FROM "MemoryLongTerm" ml
    LEFT JOIN "MemoryRelevance" mr ON ml."twinId" = mr."twinId" AND ml.key = mr."memoryKey"
    WHERE ml."twinId" = $1
    ORDER BY relevance DESC, hits DESC, ml."updatedAt" DESC
    LIMIT $2
  `, [twinId, limit]);
  
  // Update hit count when used
  for (const mem of memories.rows) {
    await this.updateMemoryRelevance(twinId, mem.key, mem.relevance + 0.1);
  }
  
  return memories.rows;
}
```

Impact: Better memory prioritization

```typescript
// New service: cacheService.ts
import Redis from 'ioredis';

const redis = new Redis(process.env.REDIS_URL);

export async function getCachedPrompt(twinId: string, contextHash: string): Promise<string | null> {
  const key = `twin:${twinId}:prompt:${contextHash}`;
  return await redis.get(key);
}

export async function setCachedPrompt(twinId: string, contextHash: string, prompt: string, ttl: number = 3600): Promise<void> {
  const key = `twin:${twinId}:prompt:${contextHash}`;
  await redis.setex(key, ttl, prompt);
}

// In promptBuilder.ts
async buildSystemPrompt(context: PromptContext): Promise<string> {
  // Create context hash
  const contextHash = crypto.createHash('md5')
    .update(JSON.stringify({
      twinId: context.twinId,
      sessionMemory: context.sessionMemory?.summary,
      messageCount: context.chatMemory.length
    }))
    .digest('hex');
  
  // Check cache
  const cached = await getCachedPrompt(context.twinId!, contextHash);
  if (cached) return cached;
  
  // Build prompt (existing logic)
  const prompt = await this.assemblePrompt(...);
  
  // Cache for 1 hour
  await setCachedPrompt(context.twinId!, contextHash, prompt, 3600);
  
  return prompt;
}
```

Impact: 10x faster prompt assembly (500ms → 50ms)  
Effort: Medium (3-4 hours) + Redis setup

```typescript
// New service: personaValidator.ts
export async function validatePersonaConsistency(twinId: string): Promise<boolean> {
  // Get current style vector
  const twin = await db.query('SELECT "styleVector", "personaData" FROM "Twin" WHERE id = $1', [twinId]);
  
  // Get all style anchors
  const anchors = await db.query('SELECT * FROM style_anchors WHERE twin_id = $1', [twinId]);
  
  // Analyze consistency using OpenAI
  const consistencyCheck = await openai.chat.completions.create({
    model: 'gpt-4o-mini',
    messages: [{
      role: 'system',
      content: 'Analyze if style anchors are consistent with style vector. Return JSON: {consistent: boolean, driftScore: number}'
    }, {
      role: 'user',
      content: `Style Vector: ${JSON.stringify(twin.styleVector)}\nAnchors: ${JSON.stringify(anchors)}`
    }]
  });
  
  const result = JSON.parse(consistencyCheck.choices[0]?.message?.content || '{}');
  
  if (result.driftScore > 0.7) {
    // Regenerate base prompt
    await regenerateSystemPrompt(twinId);
    return false;
  }
  
  return true;
}

// Add to cron job (weekly)
cron.schedule('0 0 * * 0', async () => {
  const twins = await db.query('SELECT id FROM "Twin" WHERE "isPublic" = true');
  for (const twin of twins.rows) {
    await validatePersonaConsistency(twin.id);
  }
});
```

Impact: Maintains twin consistency
